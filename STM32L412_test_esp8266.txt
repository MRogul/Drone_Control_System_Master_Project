  /* Initialize all configured peripherals */
  MX_GPIO_Init();
  MX_USART2_UART_Init();
  MX_CRC_Init();
  MX_USART1_UART_Init();
  /* USER CODE BEGIN 2 */

  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  memset(rx_esp32_data, 0x00, ESP32_MSG_LENGTH); //Zeruje bufor rx_esp32_data
  HAL_UART_Receive_IT(ESP32_UART_HANDLE, rx_esp32_data,ESP32_MSG_LENGTH); // uruchamia odbieranie danych przez UART w trybie przerwań. Kiedy dane przyjdą, zostanie wywołane HAL_UART_RxCpltCallback.
  while (1) {
      if (esp32_data_received_flag == 1) {
          esp32_data_received_flag = 0;
          uint8_t received_crc = rx_esp32_data[5];

          // Oblicz CRC dla pierwszych 5 bajtów (używając HAL CRC)
          uint32_t computed_crc32 = HAL_CRC_Calculate(&hcrc, (uint32_t*)rx_esp32_data, 5);
          uint8_t computed_crc = (uint8_t)(computed_crc32 & 0xFF); // Weź tylko najniższy bajt

          if (computed_crc == received_crc) {
              int32_t value;
              memcpy(&value, &rx_esp32_data[1], 4);  // Odczytaj 4-bajtową wartość
              switch (rx_esp32_data[0]) {
                  case 0x01: kp = PID_KP_MIN + (PID_KP_MAX - PID_KP_MIN) * value / 100.0f; break;
                  case 0x02: ki = PID_KI_MIN + (PID_KI_MAX - PID_KI_MIN) * value / 100.0f; break;
                  case 0x03: kd = PID_KD_MIN + (PID_KD_MAX - PID_KD_MIN) * value / 100.0f; break;
                  case 0x04: tau = PID_TAU_MIN + (PID_TAU_MAX - PID_TAU_MIN) * value / 100.0f; break;
                  default: break;
              }
          }
      }
  }
    /* USER CODE END WHILE */

    /* USER CODE BEGIN 3 */

  /* USER CODE END 3 */
}
